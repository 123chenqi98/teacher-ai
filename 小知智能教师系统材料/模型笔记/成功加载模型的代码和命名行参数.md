运行代码：optimized_training.py

```
import sys
import os

# 添加项目路径
sys.path.insert(0, os.path.join(os.path.dirname(__file__), 'src'))

from llamafactory.train.tuner import run_exp


def main():
    # 设置环境变量优化CUDA内存管理
    os.environ['PYTORCH_CUDA_ALLOC_CONF'] = 'expandable_segments:True'

    # 核心补充：添加stage和do_train，同时修正参数兼容性
    args = [
        '--stage', 'sft',  # 必须：指定训练阶段为监督微调
        '--do_train', 'True',  # 必须：显式开启训练
        '--model_name_or_path', 'E:/EduChat/models/educhat-r1-001-8b-qwen3.0',
        '--template', 'qwen',  # Qwen3.0专用模板
        '--dataset', 'educhat_custom',  # 注意：需确保data目录下有该数据集的配置/文件
        '--dataset_dir', 'E:/EduChat/LLaMA-Factory/data',
        '--finetuning_type', 'lora',
        '--output_dir', 'E:/EduChat/finetuned_model',
        '--per_device_train_batch_size', '1',
        '--gradient_accumulation_steps', '16',
        '--num_train_epochs', '1',
        '--learning_rate', '5e-5',
        '--logging_steps', '10',
        '--save_steps', '100',
        '--warmup_steps', '10',
        '--lora_rank', '8',
        '--lora_alpha', '16',
        '--lora_dropout', '0.05',
        '--fp16',  # 启用混合精度训练
        '--quantization_bit', '8',  # 简化：直接用quantization_bit，无需指定bnb（默认就是）
        '--low_cpu_mem_usage', 'True',
        '--cutoff_len', '2048',  # Qwen3.0建议的序列长度
        '--lora_target', 'q_proj,v_proj,k_proj,o_proj',  # Qwen3.0 LoRA核心模块
        '--do_eval', 'False',  # 禁用验证，加快训练
        '--optim', 'adamw_bnb_8bit',  # 适配8bit量化的优化器
        '--report_to', 'none',  # 关闭WANDB
        '--trust_remote_code', 'True',  # Qwen3.0必须
        '--gradient_checkpointing', 'True',  # 节省显存
    ]

    # 运行训练
    run_exp(args)


if __name__ == "__main__":
    main()
```

命名行指令和参数：python optimized_training.py --stage sft --do_train True --model_name_or_path E:/EduChat/models/educhat-r1-001-8b-qwen3.0 --dataset educhat_train.jsonl --dataset_format jsonl --template qwen --finetuning_type lora --lora_rank 8 --per_device_train_batch_size 4 --learning_rate 2e-4 --num_train_epochs 3 --cutoff_len 2048 --output_dir ./output/qwen-8b-sft --quantization_bit 8 --trust_remote_code True --gradient_checkpointing True --report_to none --optim adamw_bnb_8bit
