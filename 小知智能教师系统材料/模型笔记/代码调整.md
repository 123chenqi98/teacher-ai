#### 2. 训练配置优化

bash运行

```bash
# 优化后的训练命令（关键调整）
python optimized_training.py \
  --stage sft \
  --do_train True \
  --model_name_or_path E:/EduChat/models/educhat-r1-001-8b-qwen3.0 \
  --dataset educhat_train.jsonl \
  --dataset_format jsonl \
  --template qwen \
  --finetuning_type lora \
  --lora_rank 16 \          # 提升LoRA秩，增强拟合能力
  --lora_target q_proj,v_proj,k_proj,o_proj \  # 明确Qwen的LoRA目标层
  --per_device_train_batch_size 2 \            # 降低batch_size，避免OOM
  --gradient_accumulation_steps 8 \            # 梯度累积，总batch_size=16
  --learning_rate 1e-4 \                       # 降低学习率，更稳定
  --num_train_epochs 10 \                      # 增加epoch（数据充足时）
  --cutoff_len 4096 \                          # 适配Qwen的长上下文
  --output_dir ./output/qwen-8b-sft \
  --quantization_bit 4 \                       # 4bit量化，节省显存（8bit可保留）
  --trust_remote_code True \
  --gradient_checkpointing True \
  --report_to none \
  --optim adamw_bnb_8bit \
  --upcast_layernorm True \                    # 解决量化训练的layernorm警告
  --flash_attn auto \                          # 自动启用FlashAttention
```
